# Bank Account class
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount
        print(f"Joma: Tk {amount}. Balance: Tk {self.balance}")

    def withdraw(self, amount):
        if amount > self.balance:
            print("Taka nei!")
        else:
            self.balance -= amount
            print(f"Tola: Tk {amount}. Balance: Tk {self.balance}")

acc = BankAccount("Rahim", 1000)
acc.deposit(500)
acc.withdraw(200)

    print(f"\n--- Round {r}/5 ---")

print(f"\nFinal: You {p_score} - Computer {c_score}")

🚀 Project 6: Contact Book (with File Save)

import json

def load():
    try:
        with open("contacts.json","r") as f: return json.load(f)
    except: return {}

def save(contacts):
    with open("contacts.json","w") as f: json.dump(contacts,f,indent=2)

contacts = load()
while True:
    print("\n📱 1.Show 2.Add 3.Search 4.Delete 5.Exit")
    ch = input("Choice: ")
    if ch=="1":
        for n,num in contacts.items(): print(f"  {n}: {num}")
    elif ch=="2":
        n=input("Name: "); num=input("Number: ")
        contacts[n]=num; save(contacts); print("✅ Added!")
    elif ch=="3":
        s=input("Search: ")
        for n,num in contacts.items():
            if s.lower() in n.lower(): print(f"  {n}: {num}")
    elif ch=="4":
        n=input("Delete name: ")
        if n in contacts: del contacts[n]; save(contacts); print("🗑 Deleted!")
    elif ch=="5": print("Bye!"); break

import json

class StudentManager:
    def __init__(self):
        self.students = self.load()

    def load(self):
        try:
            with open("students.json","r") as f: return json.load(f)
        except: return []

    def save(self):
        with open("students.json","w") as f:
            json.dump(self.students, f, indent=2)

    def grade(self, m):
        if m>=80: return "A+"
        elif m>=70: return "A"
        elif m>=60: return "B"
        elif m>=50: return "C"
        else: return "F"

    def add(self):
        n = input("Name: ")
        r = input("Roll: ")
        m = float(input("Marks: "))
        self.students.append({"name":n,"roll":r,"marks":m,"grade":self.grade(m)})
        self.save()
        print(f"✅ {n} added!")

    def show(self):
        if not self.students: print("No students!"); return
        print(f"{'Name':<15}{'Roll':<10}{'Marks':<10}{'Grade':<5}")
        print("-"*40)
        for s in self.students:
            print(f"{s['name']:<15}{s['roll']:<10}{s['marks']:<10}{s['grade']:<5}")

    def search(self):
        q = input("Search: ")
        found = [s for s in self.students
                 if q.lower() in s['name'].lower() or q==s['roll']]
        for s in found:
            print(f"{s['name']} - Roll:{s['roll']} - Marks:{s['marks']}")
        if not found: print("Not found!")

    def stats(self):
        if not self.students: print("No data!"); return
        marks = [s['marks'] for s in self.students]
        print(f"Students: {len(self.students)}")
        print(f"Average: {sum(marks)/len(marks):.1f}")
        print(f"Highest: {max(marks)}")
        print(f"Lowest: {min(marks)}")

mgr = StudentManager()
while True:
    print("\n🎓 1.Show 2.Add 3.Search 4.Stats 5.Exit")
    ch = input("Choice: ")
    if ch=="1": mgr.show()
    elif ch=="2": mgr.add()
    elif ch=="3": mgr.search()
    elif ch=="4": mgr.stats()
    elif ch=="5": print("Bye!"); break
